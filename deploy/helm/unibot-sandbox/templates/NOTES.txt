The UserSandbox CRD and operator are installed.

  UNIBOT_SANDBOX_DRIVER=kubernetes
  UNIBOT_SANDBOX_KUBERNETES_NAMESPACE={{ .Release.Namespace }}
  UNIBOT_SANDBOX_RUNTIME_CLASS={{ .Values.runtimeClass.name }}
  UNIBOT_SANDBOX_DEFAULT_IMAGE={{ .Values.sandbox.image.repository }}:{{ .Values.sandbox.image.tag }}

  UNIBOT_SANDBOX_KUBERNETES_WORKSPACE_PVC=<pre-created-rwx-pvc>

Verify:
